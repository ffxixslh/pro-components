# RButton

## 基本使用

```tsx preview
import { RButton } from "@ffxixslh/pro-components";
import { sleep } from "@ffxixslh/utils";

export default () => {
  return (
    <RButton
      key={"reload"}
      onClick={async () => {
        await sleep(1000);
      }}
    >
      刷新
    </RButton>
  );
};

```
